استقرار · داکر

یک فرمان برای تولدِ یک تمدن

Docker Compose کل استک را بالا می‌آورد: PostgreSQL، Redis، بک‌اند، فرانت‌اند — یک فرمان، چهار سرویس.

docker-compose.yml
version: '3.9'

services:
  postgres:
    image: postgres:15-alpine
    environment:
      POSTGRES_DB: fanus
      POSTGRES_USER: fanus
      POSTGRES_PASSWORD: ${DB_PASS}
    volumes:
      - pg_data:/var/lib/postgresql/data
    ports: ["5432:5432"]
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U fanus"]
      interval: 10s
      timeout: 5s
      retries: 5

  redis:
    image: redis:7-alpine
    ports: ["6379:6379"]
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s

  backend:
    build: ./backend
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    environment:
      DATABASE_URL: postgresql+asyncpg://fanus:${DB_PASS}@postgres/fanus
      REDIS_URL: redis://redis:6379
      JWT_SECRET: ${JWT_SECRET}
      FANUS_ENGINE_PATH: /app/fanus-engine
    ports: ["8000:8000"]
    command: uvicorn app.main:app --host 0.0.0.0 --port 8000 --reload
    volumes:
      - ./backend:/app
      - ./fanus-engine:/app/fanus-engine

  frontend:
    build: ./frontend
    depends_on: [backend]
    ports: ["3000:3000"]
    environment:
      VITE_API_URL: http://localhost:8000
      VITE_WS_URL: ws://localhost:8000
    volumes:
      - ./frontend:/app
      - /app/node_modules

volumes:
  pg_data:
